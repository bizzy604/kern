function tbl(cols, headers, rows2) {
  return new Table({ width:{size:W,type:WidthType.DXA}, columnWidths:cols,
    rows:[row(headers.map((h,i)=>hCell(h,cols[i]))),
      ...rows2.map((r,ri)=>row(r.map((c,i)=> typeof c==='string'
        ? (i===0?cellB(c,cols[i],ri%2===0?LGRAY:WHITE):cell(c,cols[i],ri%2===0?LGRAY:WHITE))
        : c)))]});
}
